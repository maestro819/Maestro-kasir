// ========================================== // 1. KONFIGURASI SUPABASE & DATABASE LOKAL // ========================================== // GANTI 2 BARIS DI BAWAH INI DENGAN DATA DARI SUPABASE ANDA const SUPABASE_URL = 'https://URL_SUPABASE_ANDA.supabase.co'; const SUPABASE_KEY = 'ANON_KEY_SUPABASE_ANDA'; // Inisialisasi Supabase Client const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_KEY); // Inisialisasi Database Lokal Dexie.js (Offline Storage) const db = new Dexie("MaestroPayDB"); db.version(1).stores({ products: 'id, name, price, stock, barcode', transactions: 'id, sync_status' // sync_status: 'pending' atau 'synced' }); // ========================================== // 2. LOGIKA AUTENTIKASI (SUPABASE AUTH) // ========================================== // Process Login dengan Email & Password async function prosesLogin() { const email = document.getElementById('login-email').value.trim(); const password = document.getElementById('login-password').value; if (!email || !password) { alert("Email dan Password wajib diisi!"); return; } const { data, error } = await supabaseClient.auth.signInWithPassword({ email: email, password: password }); if (error) { alert("Login Gagal: " + error.message); } else { tampilkanAplikasi(data.user.email); } } // Process Logout async function prosesLogout() { await supabaseClient.auth.signOut(); window.location.reload(); } // Cek Sesi Login Saat Halaman Dibuka/Refresh async function cekSesiAktif() { const { data: { session } } = await supabaseClient.auth.getSession(); if (session) { tampilkanAplikasi(session.user.email); } else { document.getElementById('login-container').style.display = 'block'; document.getElementById('app-container').style.display = 'none'; } updateOfflineUI(); } function tampilkanAplikasi(email) { document.getElementById('login-container').style.display = 'none'; document.getElementById('app-container').style.display = 'block'; const emailLabel = document.getElementById('user-email-display'); if (emailLabel) emailLabel.innerText = "Terhubung sebagai: " + email; // Muat data produk begitu berhasil masuk syncProductsToLocal(); } // ========================================== // 3. LOGIKA PRODUK (SINKRONISASI LOKAL/SERVER) // ========================================== async function syncProductsToLocal() { // 1. Jika Sedang Offline: Gunakan data lokal dari Dexie if (!navigator.onLine) { const localProducts = await db.products.toArray(); renderProduk(localProducts); return; } // 2. Jika Sedang Online: Tarik dari Supabase (Row Level Security / RLS aktif secara otomatis) const { data, error } = await supabaseClient.from('products').select('*'); if (error) { console.error("Gagal mengambil produk dari Supabase:", error.message); return; } // 3. Perbarui database lokal Dexie await db.products.clear(); if (data.length > 0) { await db.products.bulkAdd(data); } renderProduk(data); } function renderProduk(products) { const container = document.getElementById('daftar-produk'); container.innerHTML = ''; if (!products || products.length === 0) { container.innerHTML = '

Belum ada data produk di toko ini.

'; return; } products.forEach(p => { container.innerHTML += `

${p.name}

Stok: ${p.stock || 0}

Rp ${(p.price || 0).toLocaleString('id-ID')}
`; }); } // ========================================== // 4. TRANSAKSI (OFFLINE-FIRST) // ========================================== // Simulasi Kasir Melakukan Checkout async function simulasiCheckout() { const txData = { id: Date.now(), // Unique ID menggunakan Timestamp total: 50000, bayar: 50000, kembali: 0, metode: 'Tunai', items: [{ name: 'Barang Contoh', qty: 1, price: 50000 }], sync_status: 'pending' // Status penanda belum terkirim ke Supabase }; // 1. SIMPAN KE DATABASE LOKAL TERLEBIH DAHULU (Sangat Cepat & Tidak Membutuhkan Internet) await db.transactions.put(txData); alert("Transaksi berhasil disimpan secara lokal!"); // 2. Coba kirim data ke Supabase di latar belakang triggerBackgroundSync(); } // Background Sync: Kirim Transaksi Lokal ke Supabase Saat Online async function triggerBackgroundSync() { if (!navigator.onLine) { updateOfflineUI(); return; } // Ambil semua transaksi lokal yang statusnya masih 'pending' const pendingTx = await db.transactions.where('sync_status').equals('pending').toArray(); if (pendingTx.length === 0) { updateOfflineUI(); return; } try { // Sesuaikan payload sesuai dengan kolom tabel 'transactions' Supabase const payload = pendingTx.map(tx => ({ id: tx.id, total: tx.total, bayar: tx.bayar, kembali: tx.kembali, metode: tx.metode, items: tx.items })); // Insert ke Supabase const { error } = await supabaseClient.from('transactions').insert(payload); if (!error) { // Ubah status lokal dari 'pending' menjadi 'synced' const syncedIds = pendingTx.map(tx => tx.id); await db.transactions.where('id').anyOf(syncedIds).modify({ sync_status: 'synced' }); } } catch (err) { console.error("Gagal sinkronisasi transaksi ke server:", err); } updateOfflineUI(); } // ========================================== // 5. EVENT LISTENERS & STATUS KONEKSI // ========================================== async function updateOfflineUI() { const indicator = document.getElementById('offline-indicator'); if (!indicator) return; const pendingCount = await db.transactions.where('sync_status').equals('pending').count(); const countSpan = document.getElementById('pending-count'); if (countSpan) countSpan.innerText = pendingCount; if (!navigator.onLine || pendingCount > 0) { indicator.style.display = 'block'; if (navigator.onLine) { indicator.className = "bg-yellow-500 text-white text-center p-1 text-sm font-bold z-50 fixed top-0 w-full shadow"; indicator.innerHTML = `⚠️ Menyinkronkan ${pendingCount} transaksi tertunda ke server...`; } else { indicator.className = "bg-red-500 text-white text-center p-1 text-sm font-bold z-50 fixed top-0 w-full shadow"; indicator.innerHTML = `🔴 Mode Offline - Transaksi Tertunda: ${pendingCount}`; } } else { indicator.style.display = 'none'; } } // Deteksi otomatis status koneksi internet dan inisialisasi aplikasi window.addEventListener('load', cekSesiAktif); window.addEventListener('online', () => { updateOfflineUI(); triggerBackgroundSync(); }); window.addEventListener('offline', updateOfflineUI);
